cmake_minimum_required(VERSION 3.18)
project(libri_pr17_microbenchmark LANGUAGES CXX)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(OpenMP REQUIRED COMPONENTS CXX)
find_package(CUDAToolkit REQUIRED)
find_package(BLAS REQUIRED)
find_package(LAPACK REQUIRED)
find_library(TEST_DDLA NAMES ddla PATHS "${DDLA_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
foreach(scalar real complex)
  add_executable(benchmark_${scalar} driver.cpp)
  target_compile_features(benchmark_${scalar} PRIVATE cxx_std_17)
  target_compile_definitions(benchmark_${scalar} PRIVATE __GPU_RI __DDLA_RI DDLA_USE_CUDA)
  if(scalar STREQUAL complex)
    target_compile_definitions(benchmark_${scalar} PRIVATE TEST_COMPLEX)
  endif()
  if(BENCH_RESIDENCY)
    target_compile_definitions(benchmark_${scalar} PRIVATE BENCH_RESIDENCY)
  endif()
  target_include_directories(benchmark_${scalar} PRIVATE "${LIBRI_ROOT}/include" "${LIBCOMM_ROOT}/include" "${CEREAL_ROOT}/include" "${DDLA_ROOT}/include")
  target_link_libraries(benchmark_${scalar} PRIVATE MPI::MPI_CXX OpenMP::OpenMP_CXX ${LAPACK_LIBRARIES} ${BLAS_LIBRARIES} ${TEST_DDLA} CUDA::cudart CUDA::cublas CUDA::curand CUDA::cusolver)
endforeach()
